\begin{slide}

	\slidetitle{File service model}
	
	\begin{minipage}[t]{0.45\textwidth}
	\textbf{Upload/Download model}
	\begin{itemize}
		\item Read file: copy file from server to client
		\item Write file: copy file from client to server
		\item Advantage: 
		\begin{itemize}
			\item Simple
		\end{itemize}
		\item Problem: 
		\begin{itemize}
			\item Wasteful: what if client needs small piece?
			\item Problematic: what if client doesn’t have enough space?
			\item Consistency: what if others need to modify the same file?
		\end{itemize}
	\end{itemize}
	\end{minipage}
	\hfill
	\begin{minipage}[t]{0.45\textwidth}
	\textbf{Remote access model}
	\begin{itemize}
		\item File service provides functional interface
		\begin{itemize}
			\item create, delete, read bytes, write bytes, etc…
		\end{itemize}
		\item Advantage: 
		\begin{itemize}
			\item Client gets only what’s needed
			\item Server can manage coherent view of file system
		\end{itemize}
		\item Problem: 
		\begin{itemize}
			\item Possible server and network congestion
			\item Servers are accessed for duration of file access
			\item Same data may be requested repeatedly
		\end{itemize}
	\end{itemize}
	\end{minipage}

\end{slide}

\begin{slide}
	
	\slidetitle{Semantics of file sharing} 
	
	\begin{minipage}[t]{0.45\textwidth}
	\textbf{Sequential Semantics}
	\medskip
	
	Read returns result of last write
	\medskip
	
	Easily achieved if
	\begin{itemize}
		\item Only one server
		\item Clients do not cache data
	\end{itemize}
	\end{minipage}
	\hfill
	\begin{minipage}[t]{0.45\textwidth}
	\textbf{Session Semantics}
	\medskip
	
	Relax the rules
	\begin{itemize}
		\item Changes to an open file are initially visible only to the process (or machine) that modified it.
		\item Need to hide or lock file under modification from other clients
		\item Last process to modify the file wins.
	\end{itemize}
	\end{minipage}
	
\end{slide}

\begin{slide}

	\slidetitle{Stateful or Stateless design?}
	
	\textbf{Stateful}: Server maintains client-specific state
	\begin{itemize}
		\item File locking is possible
		\item Better performance in processing requests
	\end{itemize}
	\bigskip
	
	\textbf{Stateless}: Server maintains no information on client accesses
	\begin{itemize}
		\item Each request must identify file and offsets
		\item Server and Client can crash and recover		
		\item Problem if file is deleted on server
	\end{itemize}
	
\end{slide}

\begin{slide}

	\slidetitle{Approaches to caching}
	
	\textbf{Write-through}
	\begin{itemize}
		\item What if another client reads its own (out-of-date) cached copy?
	\end{itemize}
	\medskip
	
	\textbf{Delayed writes}
	\begin{itemize}
		\item Data can be buffered locally (watch out for consistency – others won’t see updates!)
		\item Remote files updated periodically
	\end{itemize}
	\medskip
	
	\textbf{Write on close}
	\begin{itemize}
		\item Admit that we have session semantics.
	\end{itemize}
	\medskip
	
	\textbf{Centralized control}
	\begin{itemize}
		\item Keep track of who has what open and cached on each node.
		\item Stateful file system with signaling traffic.
	\end{itemize}
	
\end{slide}

\begin{slide}

	\slidetitle{NFS transport protocol}
	
	For the first few versions NFS ran over UDP using Sun Remote Procedure Calls.
	\bigskip
	
	Why was UDP chosen?
	\begin{itemize}
		\item Slightly faster than TCP
		\item No connection to maintain (or lose)
		\item NFS is designed for Ethernet LAN environment - relatively reliable
		\item UDP has error detection (drops bad packets) but no retransmission
			\begin{itemize}
				\item NFS retries lost RPC requests
			\end{itemize}
	\end{itemize}
	
\end{slide}

\begin{slide}

	\slidetitle{NFS Performance}
	
	Usually slower than local
	\bigskip
	
	Improve by caching at client
	\begin{itemize}
		\item Goal: reduce number of remote operations
		\item Cache results of read, readlink, getattr, lookup, readdir
		\item Cache file data at client (buffer cache)
		\item cache file attribute information at client
		\item Cache pathname bindings for faster lookups
	\end{itemize}
	\bigskip
	
	Server side
	\begin{itemize}
		\item Caching is "automatic" via buffer cache
		\item All NFS writes are write-through to disk to avoid unexpected data loss if server dies
	\end{itemize}

\end{slide}

\begin{slide}
	
	\slidetitle{Inconsistencies may arise}
	
	Try to resolve by validation
	\begin{itemize}
		\item Save timestamp of file
		\item When file opened or server contacted for new block
		\begin{itemize}
			\item Compare last modification time
			\item If remote is more recent, invalidate cached data
		\end{itemize}
	\end{itemize}
	\bigskip
	
	Always invalidate data after some time
	\begin{itemize}
		\item After 3 seconds for open files (data blocks)
		\item After 30 seconds for directories
	\end{itemize}
	\bigskip
	
	If data block is modified, it is:
	\begin{itemize}
		\item Marked dirty
		\item Scheduled to be written
		\item Flushed on file close
	\end{itemize}
\end{slide}

\begin{slide}

	\slidetitle{Problems with NFS}
	
	File consistency
	\medskip
	
	Assumes clocks are synchronized
	\medskip
	
	Open with append cannot be guaranteed to work
	\medskip
	
	Locking cannot work
	\begin{itemize}
		\item Separate lock manager added (but this adds stateful behavior)
	\end{itemize}
	\medskip
	
	No reference counting of open files
	\begin{itemize}
		\item You can delete a file you (or others) have open!
	\end{itemize}

\end{slide}

\begin{slide}

	\slidetitle{Improving NFS: version 2}
	
	User-level lock manager
	\begin{itemize}
		\item Monitored locks: introduces state at server 
		\item (but runs as a separate user-level process) 
		\begin{itemize}
			\item If server crashes: status monitor reinstates locks on recovery
			\item If client crashes: all locks from client are freed 
		\end{itemize}
	\end{itemize}
	\medskip
	
	NV RAM support 
	\begin{itemize}
		\item Improves write performance 
		\item Normally NFS must write to disk on server before responding to client write requests 
		\item Relax this rule through the use of non-volatile RAM 
	\end{itemize}
	

\end{slide}

\begin{slide}

	\slidetitle{Improving NFS: version 2}
	
	Adjust RPC retries dynamically
	\begin{itemize}
		\item Reduce network congestion from excess RPC retransmissions under load 
		\item Based on performance
	\end{itemize}
	\medskip
	
	Client-side disk caching
	\begin{itemize}
		\item cacheFS
		\item Extend buffer cache to disk for NFS
	\end{itemize}

\end{slide}

\begin{slide}

	\slidetitle{Improving NFS: version 3}
	
	Support 64-bit file sizes
	\medskip
	
	TCP support and large-block transfers
	\medskip
	
	Negotiate for optimal transfer size
	\medskip
	
	Server checks access for entire path from client
	
\end{slide}

\begin{slide}

	\slidetitle{Improving NFS: version 3}
	
	New commit operation
	\begin{itemize}
		\item Check with server after a write operation to see if data is committed 
		\item If commit fails, client must resend data 
		\item Reduces number of write requests to server 
		\item Speeds up write requests by NOT requiring server to write to disk immediately
	\end{itemize}
	\medskip
	
	Return file attributes with each request
	
\end{slide}

\end{slide}

\begin{slide}

	\slidetitle{AFS Assumptions}
	
	Most files are small 
	\medskip
	
	Reads are more common than writes 
	\medskip
	
	Most files are accessed by one user at a time
	\medskip
	
	Files are referenced in bursts (locality)
	\begin{itemize}
		\item Once referenced, a file is likely to be referenced again 
	\end{itemize}
	
\end{slide}

\begin{slide}

	\slidetitle{AFS Design Decisions}
	
	Whole file serving
	\begin{itemize}
		\item Send the entire file on open 
	\end{itemize}
	\medskip
	
	Whole file caching
	\begin{itemize}
		\item Client caches entire file on local disk 
		\item Client writes the file back to server on close 
		\begin{itemize} 
			\item if modified 
			\item Keeps cached copy for future accesses
		\end{itemize}
	
	\end{itemize}
	
\end{slide}

\begin{slide}

	\slidetitle{AFS Design}
	
	Each client has an AFS disk cache
	\begin{itemize}
		\item Part of disk devoted to AFS (e.g. 100 MB)
		\item Client manages cache in LRU manner
	\end{itemize}
	\medskip
	
	Clients communicate with set of trusted servers 
	\medskip
	
	Each server presents one identical name space to clients
	\begin{itemize}
		\item All clients access it in the same way 
		\item Location transparent
	\end{itemize}
	
\end{slide}

\begin{slide}

	\slidetitle{AFS cache coherence}
	
	On open: 
	\begin{itemize}
		\item Server sends entire file to client and provides a callback promise: 
		\item It will notify the client when any other process modifies the file 
	\end{itemize}
	\medskip
	
	If a client modified a file: 
	\begin{itemize}
		\item Contents are written to server on close 
	\end{itemize}
	\medskip
	
	When a server gets an update: 
	\begin{itemize}
		\item it notifies all clients that have been issued the callback promise 
		\item Clients invalidate cached files 
	\end{itemize}
	
\end{slide}

\begin{slide}

	\slidetitle{AFS cache coherence}
	
	If a client was down  
	\begin{itemize}
		\item On startup, contact server with timestamps of all cached files to decide whether to invalidate If a process has a file open
		\item It continues accessing it even if it has been invalidate
		\item Upon close, contents will be propagated to server 
	\end{itemize}
	\medskip
	
	If a process has a file open
	\begin{itemize}
		\item It continues accessing it even if it has been invalidate 
		\item Upon close, contents will be propagated to server 
	\end{itemize}
	
\end{slide}
